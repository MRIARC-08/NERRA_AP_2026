-- Q12: Detect possible attribute contradictions between active claims.
-- This file contains the query to explain, not a fabricated execution plan.
-- After loading the documented test dataset, run:
-- EXPLAIN (ANALYZE, BUFFERS, VERBOSE, SETTINGS)
-- followed by the SELECT below, then save the actual terminal output here.

SELECT
    c1.id AS claim_a_id,
    c1.subject,
    c1.attribute,
    c1.value AS value_a,
    c1.chapter AS chapter_a,
    c1.evidence_quote AS evidence_a,
    c2.id AS claim_b_id,
    c2.value AS value_b,
    c2.chapter AS chapter_b,
    c2.evidence_quote AS evidence_b
FROM claims AS c1
JOIN claims AS c2
  ON c1.project_id = c2.project_id
 AND LOWER(c1.subject) = LOWER(c2.subject)
 AND LOWER(c1.attribute) = LOWER(c2.attribute)
 AND c1.id < c2.id
 AND LOWER(c1.value) <> LOWER(c2.value)
WHERE c1.project_id = '00000000-0000-0000-0000-000000000002'
  AND c1.status = 'active'
  AND c2.status = 'active';
